CREATE OR REPLACE FUNCTION trace(
    logname text,
    when_time text,
    row_nature text,
    stack_depth text,
    function_name text,
    content text
)
RETURNS void AS $$
DECLARE
    table_exists text;
    cmd text;
BEGIN
    -- Check if the table exists in the database
    -- (equivalent to checking OBJECT_ID('Logging..' + @logname) IS NOT NULL)
    SELECT TO_REGCLASS(logname) INTO table_exists;

    IF table_exists IS NOT NULL THEN
        -- Construct the dynamic INSERT statement safely using %I for table name 
        -- and %L for string literals (escaping quotes automatically)
        cmd := format(
            'INSERT INTO %I (when_time, row_nature, stack_depth, function_name, content) '
            'VALUES (%L, %L, %L, %L, %L)',
            logname, 
            when_time, 
            row_nature, 
            stack_depth, 
            function_name, 
            content
        );

        -- Execute the dynamic insert command
        EXECUTE cmd;
    ELSE
        -- Required table not found -> do nothing (matches your T-SQL logic)
        NULL;
    END IF;
END;
$$ LANGUAGE plpgsql;
