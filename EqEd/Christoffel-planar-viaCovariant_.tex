	\section*{Christoffel Symbols via Covariant Derivative (Polar Coordinates)}
	
	\subsection*{1. Coordinate System}
	
	We use planar polar coordinates:
	\[
	(r,\theta)
	\]
	
	Position vector:
	\[
	\mathbf{r} = r\cos\theta\,\hat{\mathbf{x}} + r\sin\theta\,\hat{\mathbf{y}}
	\]
	
	\subsection*{2. Coordinate Basis}
	
	\[
	\mathbf{e}_r = \frac{\partial \mathbf{r}}{\partial r}
	= \cos\theta\,\hat{\mathbf{x}} + \sin\theta\,\hat{\mathbf{y}}
	\]
	
	\[
	\mathbf{e}_\theta = \frac{\partial \mathbf{r}}{\partial \theta}
	= -r\sin\theta\,\hat{\mathbf{x}} + r\cos\theta\,\hat{\mathbf{y}}
	\]
	
	\subsection*{3. Metric}
	
	\[
	g_{ij} = \mathbf{e}_i \cdot \mathbf{e}_j
	\]
	
	\[
	g_{rr} = 1, \quad g_{\theta\theta} = r^2, \quad g_{r\theta}=0
	\]
	
	\subsection*{4. Dual Basis}
	
	The dual basis satisfies:
	\[
	\mathbf{e}^i \cdot \mathbf{e}_j = \delta^i_j
	\]
	
	Thus:
	
	\[
	\mathbf{e}^r = \mathbf{e}_r
	\]
	
	\[
	\mathbf{e}^\theta = \frac{1}{r^2}\mathbf{e}_\theta
	\]
	
	\subsection*{5. Key Formula}
	
	\[
	\Gamma^k_{ij}
	=
	\mathbf{e}^k \cdot \frac{\partial \mathbf{e}_i}{\partial x^j}
	\]
	
	We compute all derivatives explicitly.
	
	\subsection*{6. Derivatives of Basis Vectors}
	
	\subsubsection*{(a) $\frac{\partial \mathbf{e}_r}{\partial r}$}
	
	\[
	\frac{\partial \mathbf{e}_r}{\partial r} = 0
	\]
	
	Thus:
	\[
	\Gamma^r_{rr} = \mathbf{e}^r\cdot 0 = 0
	\quad
	\Gamma^\theta_{rr} = \mathbf{e}^\theta\cdot 0 = 0
	\]
	
	\subsubsection*{(b) $\frac{\partial \mathbf{e}_r}{\partial \theta}$}
	
	\[
	\frac{\partial \mathbf{e}_r}{\partial \theta}
	= -\sin\theta\,\hat{\mathbf{x}} + \cos\theta\,\hat{\mathbf{y}}
	= \frac{1}{r}\mathbf{e}_\theta
	\]
	
	Projection:
	
	\[
	\Gamma^r_{r\theta}
	= \mathbf{e}^r \cdot \frac{1}{r}\mathbf{e}_\theta = 0
	\]
	
	\[
	\Gamma^\theta_{r\theta}
	=
	\mathbf{e}^\theta \cdot \frac{1}{r}\mathbf{e}_\theta
	=
	\frac{1}{r^2}\mathbf{e}_\theta \cdot \frac{1}{r}\mathbf{e}_\theta
	=
	\frac{1}{r^2} \cdot \frac{1}{r} \cdot r^2
	= \frac{1}{r}
	\]
	
	\subsubsection*{(c) $\frac{\partial \mathbf{e}_\theta}{\partial r}$}
	
	\[
	\frac{\partial \mathbf{e}_\theta}{\partial r}
	= -\sin\theta\,\hat{\mathbf{x}} + \cos\theta\,\hat{\mathbf{y}}
	= \frac{1}{r}\mathbf{e}_\theta
	\]
	
	Projection:
	
	\[
	\Gamma^r_{\theta r}
	= \mathbf{e}^r \cdot \frac{1}{r}\mathbf{e}_\theta = 0
	\]
	
	\[
	\Gamma^\theta_{\theta r}
	=
	\mathbf{e}^\theta \cdot \frac{1}{r}\mathbf{e}_\theta
	= \frac{1}{r}
	\]
	
	\subsubsection*{(d) $\frac{\partial \mathbf{e}_\theta}{\partial \theta}$}
	
	\[
	\frac{\partial \mathbf{e}_\theta}{\partial \theta}
	=
	-r\cos\theta\,\hat{\mathbf{x}} - r\sin\theta\,\hat{\mathbf{y}}
	= -r \mathbf{e}_r
	\]
	
	Projection:
	
	\[
	\Gamma^r_{\theta\theta}
	=
	\mathbf{e}^r \cdot (-r\mathbf{e}_r)
	= -r
	\]
	
	\[
	\Gamma^\theta_{\theta\theta}
	=
	\mathbf{e}^\theta \cdot (-r\mathbf{e}_r)
	= 0
	\]
	
	\subsection*{7. Final Result}
	
	\[
	\Gamma^r_{\theta\theta} = -r
	\]
	
	\[
	\Gamma^\theta_{r\theta}
	=
	\Gamma^\theta_{\theta r}
	= \frac{1}{r}
	\]
	
	All other Christoffel symbols vanish.
