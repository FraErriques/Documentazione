\begin{abstract}
A comprehensive mathematical note on Bernhard Riemann's 1859 derivation of the explicit formula for the prime step-function $J(x)$. We detail the Mellin inversion of $\ln \zeta(s)$, contrast the divergent integration-by-parts choice with Riemann's convergent formulation, present a rigorous proof of the vanishing boundary terms along the vertical contour, and provide an expanded, step-by-step residue calculation over the non-trivial zeros $\rho$, trivial zeros, and principal pole of $\zeta(s)$.
\end{abstract}

\section{The Mellin Inversion Setup}

Riemann defined the prime counting step-function $J(x)$ as:
\begin{equation}
    J(x) = \sum_{n=1}^{\infty} \frac{1}{n} \pi(x^{1/n}) = \pi(x) + \frac{1}{2}\pi(x^{1/2}) + \frac{1}{3}\pi(x^{1/3}) + \dots
\end{equation}
where $\pi(x)$ is normalized at jump discontinuities via $\frac{1}{2}(\pi(x^+) + \pi(x^-))$. Taking the logarithm of the Euler product for $\zeta(s)$ for $\Re(s) > 1$ yields:
\begin{equation}
    \ln \zeta(s) = \sum_{p \text{ prime}} \sum_{m=1}^{\infty} \frac{1}{m p^{ms}} = s \int_{0}^{\infty} J(x) x^{-s-1} \, dx.
\end{equation}

Dividing by $s$, the expression $\frac{\ln \zeta(s)}{s}$ is recognized as the Mellin transform of $J(x)$. Applying Perron's formula / Mellin inversion along the vertical contour $\Re(s) = a > 1$:
\begin{equation}
    J(x) = \frac{1}{2\pi i} \int_{a - i\infty}^{a + i\infty} \frac{\ln \zeta(s)}{s} x^s \, ds.
\end{equation}

\section{The Integration by Parts Strategy}

Directly attempting to evaluate integral (3) by choosing $u = \ln \zeta(s)$ and $dv = \frac{x^s}{s} ds$ requires integrating $\frac{x^s}{s}$, which generates exponential integrals $\mathrm{Ei}(s \ln x)$ inside the contour integral, leading to a divergent, intractable representation.

To circumvent this, Riemann flipped the Integration by Parts (IBP) direction on the truncated vertical segment $[a - iT, a + iT]$:
\begin{align*}
    u(s) &= \frac{\ln \zeta(s)}{s}, & dv &= x^s \, ds, \\[1ex]
    du &= \left( \frac{\zeta'(s)}{s \zeta(s)} - \frac{\ln \zeta(s)}{s^2} \right) ds, & v &= \frac{x^s}{\ln x}.
\end{align*}

Applying integration by parts yields:
\begin{equation}
    \int_{a - iT}^{a + iT} \frac{\ln \zeta(s)}{s} x^s \, ds = \left[ \frac{\ln \zeta(s)}{s} \frac{x^s}{\ln x} \right]_{a - iT}^{a + iT} - \frac{1}{\ln x} \int_{a - iT}^{a + iT} \frac{x^s}{\ln x} \, d\left( \frac{\ln \zeta(s)}{s} \right).
\end{equation}

Expanding $d\left( \frac{\ln \zeta(s)}{s} \right) = \frac{\zeta'(s)}{s \zeta(s)} ds - \frac{\ln \zeta(s)}{s^2} ds$, the contour integral decomposes into:
\begin{equation}
    J(x) = \lim_{T \to \infty} \frac{1}{2\pi i} \left[ \frac{\ln \zeta(s)}{s} \frac{x^s}{\ln x} \right]_{a - iT}^{a + iT} - \frac{1}{2\pi i \ln x} \int_{a - i\infty}^{a + i\infty} \frac{\zeta'(s)}{\zeta(s)} \frac{x^s}{s} \, ds + \frac{1}{2\pi i \ln x} \int_{a - i\infty}^{a + i\infty} \frac{\ln \zeta(s)}{s^2} x^s \, ds.
\end{equation}

\section{Vanishing of the Boundary Term}

To evaluate the limit as $T \to \infty$, consider the integrated boundary increment:
\begin{equation}
    I_B(T) = \left[ \frac{\ln \zeta(s)}{s} \frac{x^s}{\ln x} \right]_{s = a - iT}^{s = a + iT}.
\end{equation}

\begin{lemma}
For $a > 1$, $\lim_{T \to \infty} I_B(T) = 0$.
\end{lemma}

\begin{proof}
For $s = a + iT$ with $a > 1$:
\[
|\ln \zeta(a + iT)| = \left| \sum_{p} \sum_{m=1}^{\infty} \frac{1}{m p^{m(a + iT)}} \right| \le \sum_{p} \sum_{m=1}^{\infty} \frac{1}{m p^{ma}} = \ln \zeta(a) < \infty.
\]
The numerator magnitude $|x^{a + iT}| = x^a$ is constant with respect to $T$. The denominator satisfies $|s| = \sqrt{a^2 + T^2}$. Thus:
\[
|I_B(T)| \le \frac{2 x^a}{\ln x} \cdot \frac{\ln \zeta(a)}{\sqrt{a^2 + T^2}} \xrightarrow{T \to \infty} 0.
\]
\end{proof}

\section{Contour Integration and Residue Calculus for $\frac{\zeta'(s)}{\zeta(s)}$}

With the boundary term proven to vanish, the primary contribution to $J(x)$ stems from:
\begin{equation}
    J_1(x) = -\frac{1}{2\pi i \ln x} \int_{a - i\infty}^{a + i\infty} \frac{\zeta'(s)}{\zeta(s)} \frac{x^s}{s} \, ds.
\end{equation}

To compute $J_1(x)$, we employ Hadamard's product expansion for the completed zeta function $\xi(s) = \frac{1}{2} s(s-1)\pi^{-s/2}\Gamma(s/2)\zeta(s)$. Taking the logarithmic derivative gives:
\begin{equation}
    \frac{\zeta'(s)}{\zeta(s)} = -\frac{1}{s} - \frac{1}{s-1} + \frac{1}{2}\ln \pi - \frac{1}{2}\frac{\Gamma'(s/2+1)}{\Gamma(s/2+1)} + \sum_{\rho} \left( \frac{1}{s - \rho} + \frac{1}{\rho} \right),
\end{equation}
where the sum runs over all non-trivial zeros $\rho = \beta + i\gamma$ of $\zeta(s)$, ordered symmetrically by $|\gamma|$.

\subsection{Evaluation via Contour Enclosure and Residues}

We enclose the line of integration $\Re(s) = a > 1$ by a large rectangular contour $C_R$ extending into the left half-plane ($\Re(s) \to -\infty$). By Cauchy's Residue Theorem:
\begin{equation}
    \frac{1}{2\pi i} \int_{a - i\infty}^{a + i\infty} \frac{\zeta'(s)}{\zeta(s)} \frac{x^s}{s} \, ds = \sum \operatorname{Res} \left[ \frac{\zeta'(s)}{\zeta(s)} \frac{x^s}{s} \right].
\end{equation}

We compute the residues pole by pole:

\subsubsection{1. The Pole at $s = 1$}
Near $s = 1$, $\zeta(s)$ has a simple pole with residue $1$, so $\zeta(s) = \frac{1}{s-1} + O(1)$ and $\frac{\zeta'(s)}{\zeta(s)} = -\frac{1}{s-1} + O(1)$.
\[
\operatorname{Res}_{s=1} \left[ \frac{\zeta'(s)}{\zeta(s)} \frac{x^s}{s} \right] = \lim_{s \to 1} (s-1) \left(-\frac{1}{s-1}\right) \frac{x^s}{s} = -x.
\]
Multiplying by the prefactor $-\frac{1}{\ln x}$ in $J_1(x)$ gives $\frac{x}{\ln x}$. Integrating this contribution with respect to $\ln x$ yields the fundamental term:
\begin{equation}
    \mathrm{Li}(x) = \int_{0}^{x} \frac{dt}{\ln t}.
\end{equation}

\subsubsection{2. The Non-Trivial Zeros $s = \rho$}
At each non-trivial zero $s = \rho$ of multiplicity 1, $\zeta(s)$ has a simple zero, so $\frac{\zeta'(s)}{\zeta(s)} = \frac{1}{s-\rho} + O(1)$.
\[
\operatorname{Res}_{s=\rho} \left[ \frac{\zeta'(s)}{\zeta(s)} \frac{x^s}{s} \right] = \frac{x^\rho}{\rho}.
\]
Including the prefactor $-\frac{1}{\ln x}$ and integrating gives $-\mathrm{Li}(x^\rho)$. Summing over all non-trivial zeros $\rho$ yields:
\begin{equation}
    -\sum_{\rho} \mathrm{Li}(x^\rho) = -\sum_{\Im(\rho) > 0} \left( \mathrm{Li}(x^\rho) + \mathrm{Li}(x^{\bar{\rho}}) \right).
\end{equation}

\subsubsection{3. The Trivial Zeros at $s = -2n$ ($n \in \mathbb{N}$)}
The poles of $\Gamma(s/2+1)$ at $s = -2, -4, -6, \dots$ correspond to the trivial zeros of $\zeta(s)$. At $s = -2n$:
\[
\operatorname{Res}_{s=-2n} \left[ \frac{\zeta'(s)}{\zeta(s)} \frac{x^s}{s} \right] = \frac{x^{-2n}}{-2n}.
\]
Multiplying by $-\frac{1}{\ln x}$ and summing over $n \ge 1$:
\begin{equation}
    \sum_{n=1}^{\infty} \mathrm{Li}(x^{-2n}) = \int_{x}^{\infty} \frac{dt}{t(t^2 - 1) \ln t}.
\end{equation}

\subsubsection{4. The Pole at $s = 0$}
At $s = 0$, $\frac{x^s}{s}$ has a simple pole with residue $1$, while $\frac{\zeta'(0)}{\zeta(0)} = \ln(2\pi)$. Taking into account the second term in (5), $\frac{1}{2\pi i \ln x} \int \frac{\ln \zeta(s)}{s^2} x^s ds$, the contribution at $s=0$ simplifies to the constant term:
\begin{equation}
    -\ln 2.
\end{equation}

\section{The Complete Explicit Formula}

Combining all residue contributions from Section 4 yields Bernhard Riemann's celebrated explicit formula for $J(x)$:
\begin{equation}
    J(x) = \mathrm{Li}(x) - \sum_{\rho} \mathrm{Li}(x^\rho) - \ln 2 + \int_{x}^{\infty} \frac{dt}{t(t^2 - 1) \ln t}.
\end{equation}

From $J(x)$, $\pi(x)$ is subsequently recovered by Möbius inversion:
\begin{equation}
    \pi(x) = \sum_{n=1}^{\infty} \frac{\mu(n)}{n} J(x^{1/n}).
\end{equation}
